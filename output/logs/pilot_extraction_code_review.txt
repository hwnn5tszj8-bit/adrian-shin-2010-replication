# ============================================================
# Adrian & Shin (2010) - Liquidity and Leverage
# Script 04: Automated balance sheet extraction
# ============================================================

library(rvest)
library(dplyr)
library(readr)
library(stringr)
library(tibble)

setwd("~/GitHub/adrian-shin-2010-replication")

# 1. Load downloaded documents

downloads <- read_csv(
  "output/logs/sec_document_download_log.csv",
  show_col_types = FALSE
) %>%
  filter(file.exists(local_file))

# 2. Extract first numeric value from a financial row

extract_number <- function(x) {
  
  x <- as.character(x)
  
  x <- str_trim(x)
  
  x <- gsub(",", "", x, fixed = TRUE)
  
  x <- gsub("$", "", x, fixed = TRUE)
  
  x <- str_trim(x)
  
  valid <- grepl(
    "^[0-9]+(\\.[0-9]+)?$",
    x
  )
  
  values <- suppressWarnings(
    as.numeric(x[valid])
  )
  
  if (length(values) == 0) {
    return(NA_real_)
  }
  
  values[1]
}

# 3. Extract balance sheet candidates

extract_balance_sheet <- function(path) {
  
  document <- read_html(path)
  
  tables <- html_elements(document, "table")
  
  candidates <- list()
  
  for (i in seq_along(tables)) {
    
    tab <- tryCatch(
      html_table(tables[[i]]),
      error = function(e) NULL
    )
    
    if (is.null(tab)) {
      next
    }
    
    if (nrow(tab) == 0 || ncol(tab) < 2) {
      next
    }
    
    labels <- vapply(
      seq_len(nrow(tab)),
      function(j) {
        
        cells <- as.character(
          unlist(tab[j, ], use.names = FALSE)
        )
        
        cells <- cells[
          !is.na(cells) & trimws(cells) != ""
        ]
        
        if (length(cells) == 0) {
          return("")
        }
        
        str_squish(cells[1])
      },
      character(1)
    )
    
    assets_row <- which(
      grepl(
        "^total assets\\s*$",
        labels,
        ignore.case = TRUE
      )
    )
    
    liabilities_row <- which(
      grepl(
        "^total liabilities\\s*$",
        labels,
        ignore.case = TRUE
      )
    )
    
    equity_row <- which(
      grepl(
        "^total (stockholder|shareholder|stockholders|shareholders).*equity",
        labels,
        ignore.case = TRUE
      ) |
        grepl(
          "^total equity\\s*$",
          labels,
          ignore.case = TRUE
        )
    )
    
    if (
      length(assets_row) == 0 ||
      length(liabilities_row) == 0 ||
      length(equity_row) == 0
    ) {
      next
    }
    
    assets <- extract_number(
      unlist(tab[assets_row[1], ])
    )
    
    liabilities <- extract_number(
      unlist(tab[liabilities_row[1], ])
    )
    
    equity <- extract_number(
      unlist(tab[equity_row[1], ])
    )
    
    if (anyNA(c(assets, liabilities, equity))) {
      next
    }
    
    if (assets <= 0 || equity <= 0) {
      next
    }
    
    accounting_error <- abs(
      assets - liabilities - equity
    )
    
    relative_error <- accounting_error / assets
    
    candidates[[length(candidates) + 1]] <- tibble(
      table_number = i,
      total_assets = assets,
      total_liabilities = liabilities,
      book_equity = equity,
      accounting_error = accounting_error,
      relative_error = relative_error,
      leverage = assets / equity
    )
  }
  
  if (length(candidates) == 0) {
    return(tibble())
  }
  
  bind_rows(candidates)
}

# 4. Process downloaded reports

all_candidates <- list()

for (i in seq_len(nrow(downloads))) {
  
  row <- downloads[i, ]
  
  cat(
    "\nProcessing:",
    row$bank,
    row$form,
    as.character(row$report_date),
    "\n"
  )
  
  result <- tryCatch(
    extract_balance_sheet(row$local_file),
    error = function(e) {
      
      cat("ERROR:", conditionMessage(e), "\n")
      
      tibble()
    }
  )
  
  if (nrow(result) == 0) {
    
    cat("No balance sheet candidates found.\n")
    
    next
  }
  
  result <- result %>%
    mutate(
      bank = row$bank,
      form = row$form,
      report_date = row$report_date,
      accession_number = row$accession_number,
      local_file = row$local_file,
      source_url = row$source_url,
      .before = 1
    )
  
  all_candidates[[length(all_candidates) + 1]] <- result
  
  cat("Candidates found:", nrow(result), "\n")
}

# 5. Combine extraction results

if (length(all_candidates) == 0) {
  stop("No balance sheet candidates were extracted.")
}

candidates <- bind_rows(all_candidates)

# 6. Select preliminary balance sheets
# Retain candidates satisfying the accounting identity.
# Largest assets is a heuristic, not final verification.

selected <- candidates %>%
  filter(relative_error <= 0.001) %>%
  group_by(bank, accession_number) %>%
  slice_max(
    order_by = total_assets,
    n = 1,
    with_ties = FALSE
  ) %>%
  ungroup() %>%
  mutate(
    verified = FALSE,
    source_units = NA_character_,
    extraction_method = "Automated HTML extraction"
  ) %>%
  arrange(bank, report_date)

# 7. Save results

dir.create(
  "data/processed",
  recursive = TRUE,
  showWarnings = FALSE
)

write_csv(
  candidates,
  "data/processed/balance_sheet_candidates.csv"
)

write_csv(
  selected,
  "data/processed/pilot_balance_sheets.csv"
)

# 8. Display extraction results

cat("\n========================================\n")
cat("BALANCE SHEET EXTRACTION COMPLETE\n")
cat("========================================\n")

cat("Documents available:", nrow(downloads), "\n")
cat("Candidates extracted:", nrow(candidates), "\n")
cat("Balance sheets selected:", nrow(selected), "\n\n")

print(
  selected %>%
    select(
      bank,
      report_date,
      table_number,
      total_assets,
      total_liabilities,
      book_equity,
      leverage
    ),
  n = Inf,
  width = Inf
)

# 9. Identify documents requiring further investigation

missing <- downloads %>%
  anti_join(
    selected,
    by = c("bank", "accession_number")
  )

write_csv(
  missing,
  "output/logs/balance_sheet_extraction_missing.csv"
)

cat("\nDocuments requiring further investigation:\n")

print(
  missing %>%
    select(bank, form, report_date),
  n = Inf
)

library(dplyr)
library(readr)
library(tibble)

setwd("~/GitHub/adrian-shin-2010-replication")

pilot <- tribble(
  ~bank, ~report_date, ~total_assets, ~total_liabilities, ~book_equity, ~source_form, ~source_table,
  
  "Bear Stearns", "2007-11-30",
  395362, 383569, 11793,
  "10-Q", "Comparative consolidated balance sheet",
  
  "Bear Stearns", "2008-02-29",
  398995, 387099, 11896,
  "10-Q", "Consolidated balance sheet",
  
  "Goldman Sachs", "2007-11-30",
  1119796, 1076996, 42800,
  "10-K", "Table 114 / 219",
  
  "Goldman Sachs", "2008-02-29",
  1189006, 1146377, 42629,
  "10-Q", "Table 6",
  
  "Lehman Brothers", "2007-11-30",
  691277, 668577, 22700,
  "10-K", "Table 98",
  
  "Lehman Brothers", "2008-02-29",
  786035, 761203, 24832,
  "10-Q", "Table 46 - consolidated column",
  
  "Merrill Lynch", "2008-03-28",
  1042054, 1005512, 36542,
  "10-Q", "Table 125",
  
  "Morgan Stanley", "2007-11-30",
  1045409, NA_real_, 31269,
  "10-K", "Table 42",
  
  "Morgan Stanley", "2008-02-29",
  1090896, NA_real_, 33280,
  "10-Q", "Table 250"
)

# Calculate leverage and validation indicators

pilot <- pilot %>%
  mutate(
    report_date = as.Date(report_date),
    year = lubridate::year(report_date),
    quarter = lubridate::quarter(report_date),
    report_quarter = paste0(year, "Q", quarter),
    leverage = total_assets / book_equity,
    accounting_error = total_assets -
      total_liabilities - book_equity,
    accounting_check = case_when(
      is.na(total_liabilities) ~ "Not independently checked",
      abs(accounting_error) <= 1 ~ "Passed",
      TRUE ~ "Failed"
    ),
    source_units = "USD millions - preliminary",
    extraction_method = "Pilot extraction and reconciliation",
    verified = FALSE
  ) %>%
  arrange(bank, report_date)

# Calculate preliminary quarterly growth

pilot <- pilot %>%
  group_by(bank) %>%
  mutate(
    quarter_id = year * 4 + quarter,
    
    asset_growth = if_else(
      quarter_id - lag(quarter_id) == 1,
      total_assets / lag(total_assets) - 1,
      NA_real_
    ),
    
    equity_growth = if_else(
      quarter_id - lag(quarter_id) == 1,
      book_equity / lag(book_equity) - 1,
      NA_real_
    ),
    
    leverage_growth = if_else(
      quarter_id - lag(quarter_id) == 1,
      leverage / lag(leverage) - 1,
      NA_real_
    )
  ) %>%
  ungroup()

# Save reconciled pilot dataset

dir.create(
  "data/processed",
  recursive = TRUE,
  showWarnings = FALSE
)

write_csv(
  pilot,
  "data/processed/pilot_balance_sheets_reconciled.csv",
  na = ""
)

# Display results

cat("\n========================================\n")
cat("PILOT DATASET RECONCILED\n")
cat("========================================\n")

print(
  pilot %>%
    select(
      bank,
      report_quarter,
      total_assets,
      book_equity,
      leverage,
      accounting_check
    ),
  n = Inf,
  width = Inf
)

cat("\nQuarterly growth rates (%):\n")

print(
  pilot %>%
    select(
      bank,
      report_quarter,
      asset_growth,
      equity_growth,
      leverage_growth
    ) %>%
    mutate(
      across(
        ends_with("_growth"),
        ~ round(.x * 100, 2)
      )
    ),
  n = Inf,
  width = Inf
)

cat("\nTotal observations:", nrow(pilot), "\n")
cat("Banks represented:", n_distinct(pilot$bank), "\n")
cat("\nPilot dataset saved successfully.\n")
